# pytest-randomly shuffles test order every run. It was absent, which made the
# documented `-p no:randomly` flag a silent no-op and meant the suite's
# order-independence had never actually been exercised -- installing it
# immediately surfaced a leaked TURSO_DATABASE_URL that made a unit test
# perform real DNS. Keep it installed.
pytest-randomly==4.1.0

# pytest-timeout is what makes the tests/api CI job (.github/workflows/api-tests.yml)
# fail on a hang instead of hanging until the runner's own limit. That directory
# has produced a hang before (test_combat_midbattle_api.py); nothing structural
# stops the next one, and a job that never returns reports nothing.
pytest-timeout==2.4.0
